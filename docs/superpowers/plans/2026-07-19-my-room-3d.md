# "My Room" 3D Explorable Room — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a full-screen, scroll-driven first-person 3D room at `/room` that reveals aspects of Maroof through interactive Poly Pizza low-poly objects.

**Architecture:** A code-split, `ssr:false` React Three Fiber scene. Scroll position drives a first-person camera along a spline through a room dressed with 30 GLTF props. A zustand store holds UI/scene state; a data-driven object registry + content module make objects/content editable and extensible. Overlay UI (HUD, per-object panels) is plain DOM matching the site's neo-brutalist tokens. Degrades to a 2D fallback with no WebGL.

**Tech Stack:** Next.js 16 (App Router, React 19, React Compiler), `three` 0.185, `@react-three/fiber` 9, `@react-three/drei` 10, `zustand` 5, Vitest 4 + React Testing Library.

```js
// Pure helpers for resolving Poly Pizza models. No network here (testable).

export function extractModelShortIds(html) {
  const ids = new Set();
  // Matches /m/<id> and the escaped /m/<id> form.
  const re = /(?:\/m\/|\\u002Fm\\u002F)([A-Za-z0-9]{6,})/g;
  let m;
  while ((m = re.exec(html)) !== null) ids.add(m[1]);
  return [...ids];
}

export function extractModelInfo(html) {
  const nameMatch =
    html.match(/property=["']og:title["']\s+content=["']([^"']+)["']/i) ||
    html.match(/["']name["']\s*:\s*["']([^"']+)["']/);
  const glbMatch = html.match(/https:\/\/static\.poly\.pizza\/[a-f0-9-]+\.glb(?!\.br)/i);
  let license = null;
  if (/\bCC0\b|public domain/i.test(html)) license = "CC0";
  else if (/CC-?BY|creative commons attribution/i.test(html)) license = "CC-BY";
  return {
    name: nameMatch ? nameMatch[1].trim() : null,
    glbUrl: glbMatch ? glbMatch[0] : null,
    license,
  };
}

export function pickBest(candidates, { name, author }) {
  const byName = candidates.filter(
    (c) => c.name && c.name.toLowerCase() === name.toLowerCase(),
  );
  if (byName.length === 0) return null;
  if (author) {
    const byAuthor = byName.find(
      (c) => c.author && c.author.toLowerCase() === author.toLowerCase(),
    );
    if (byAuthor) return byAuthor;
  }
  return byName[0];
}
```

```js
const OUT_DIR = "public/models/room";
const CONCURRENCY = 8;

async function getText(url) {
  const res = await fetch(url, { headers: { "user-agent": "Mozilla/5.0 room-fetch" } });
  if (!res.ok) throw new Error(`GET ${url} -> ${res.status}`);
  return res.text();
}

async function mapWithConcurrency(items, fn, limit) {
  const out = [];
  let i = 0;
  const workers = Array.from({ length: limit }, async () => {
    while (i < items.length) {
      const idx = i++;
      out[idx] = await fn(items[idx], idx);
    }
  });
  await Promise.all(workers);
  return out;
}

async function buildBundleIndex(bundleUrl) {
  const html = await getText(bundleUrl);
  const ids = extractModelShortIds(html);
  const infos = await mapWithConcurrency(
    ids,
    async (id) => {
      try {
        const page = await getText(`https://poly.pizza/m/${id}`);
        const info = extractModelInfo(page);
        return { id, ...info };
      } catch {
        return { id, name: null, glbUrl: null, license: null };
      }
    },
    CONCURRENCY,
  );
  return infos.filter((x) => x.name && x.glbUrl);
}

async function downloadGlb(url, dest) {
  const res = await fetch(url);
  if (!res.ok) throw new Error(`GET ${url} -> ${res.status}`);
  await pipeline(res.body, createWriteStream(dest));
}

async function main() {

  const index = {
    furniture: await buildBundleIndex(BUNDLES.furniture),
    office: await buildBundleIndex(BUNDLES.office),
  };

  const resolved = [];
  for (const [assetId, name, bundle, author, license] of MANIFEST) {
    const best = pickBest(index[bundle], { name, author });
    if (!best) {
      console.error(`✗ NOT FOUND: ${name} (${bundle})`);
      continue;
    }
    // Wrong author = wrong model variant = wrong license. Hard fail.
    if (best.author && best.author.toLowerCase() !== author.toLowerCase()) {
      console.error(`✗ AUTHOR MISMATCH: ${name} — wanted ${author}, page says ${best.author}`);
      continue;
    }
    // MANIFEST license is ground truth (verified against bundle listings);
    // the page-parsed license is advisory only — warn, don't fail.
    if (best.license && best.license !== license) {
      console.warn(`⚠ license advisory: ${name} — manifest ${license}, page parsed ${best.license}`);
    }
    const file = `${assetId}.glb`;
    await downloadGlb(best.glbUrl, `${OUT_DIR}/${file}`);
    resolved.push({ assetId, file, author, license, source: `https://poly.pizza/m/${best.id}` });
    console.log(`✓ ${name} -> ${file}`);
  }

  if (resolved.length !== MANIFEST.length) {

  }

  id: AssetId;
  file: string;
  author: string;
  license: "CC0" | "CC-BY";
  source: string;
}

${entries}
};

export const CC_BY_CREDITS: AssetMeta[] = Object.values(ASSETS).filter(
  (a) => a.license === "CC-BY",

`;
  await writeFile("components/room/content/assets.ts", assetsTs);

  // CREDITS.md

    .map((r) => `| ${r.assetId} | ${r.author} | ${r.license} | ${r.source} |`)

CC-BY models are attributed below as required by their license; CC0 models credited as courtesy.

| Asset | Author | License | Source |
|---|---|---|---|
${rows}

}

main().catch((e) => {
  console.error(e);
  process.exit(1);
});
```

```ts
export type StationId =
  | "workstation" | "bookcase" | "polaroids" | "corner" | "whiteboard" | "phone";
export type ObjectId = StationId | "lightswitch" | "ceilingfan";
export type Quality = "high" | "med" | "low";

  phase: "loading" | "ready";
  quality: Quality;
  reducedMotion: boolean;
  lighting: "day" | "night";
  scrollOffset: number;
  requestedOffset: number | null;
  activeStation: StationId | null;
  openObject: ObjectId | null;
  visited: ReadonlySet<ObjectId>;
  setPhase(p: RoomState["phase"]): void;
  setQuality(q: Quality): void;
  setReducedMotion(v: boolean): void;
  toggleLighting(): void;
  setScrollOffset(o: number): void;
  requestJumpTo(offset: number): void;
  clearRequestedOffset(): void;
  setActiveStation(id: StationId | null): void;
  markVisited(id: ObjectId): void;
  openObjectPanel(id: ObjectId): void;
  closeObjectPanel(): void;
}

export const useRoomStore = create<RoomState>((set) => ({
  phase: "loading",
  quality: "high",
  reducedMotion: false,
  lighting: "day",
  scrollOffset: 0,
  requestedOffset: null,
  activeStation: null,
  openObject: null,
  visited: new Set<ObjectId>(),
  setPhase: (phase) => set({ phase }),
  setQuality: (quality) => set({ quality }),
  setReducedMotion: (reducedMotion) => set({ reducedMotion }),
  toggleLighting: () => set((s) => ({ lighting: s.lighting === "day" ? "night" : "day" })),
  setScrollOffset: (scrollOffset) => set({ scrollOffset }),
  requestJumpTo: (requestedOffset) => set({ requestedOffset }),
  clearRequestedOffset: () => set({ requestedOffset: null }),
  setActiveStation: (activeStation) => set({ activeStation }),
  markVisited: (id) =>
    set((s) => {
      if (s.visited.has(id)) return s;
      const visited = new Set(s.visited);
      visited.add(id);
      return { visited };
    }),
  openObjectPanel: (id) =>
    set((s) => {
      const visited = new Set(s.visited);
      visited.add(id);
      return { openObject: id, visited };
    }),
  closeObjectPanel: () => set({ openObject: null }),
}));
```

```ts
import { useEffect, useState } from "react";
import type { Quality } from "@/components/room/state/room-store";

export interface DeviceSignals {
  cores: number;
  deviceMemory?: number;
  coarsePointer: boolean;
  dpr: number;
}

export function computeQualityTier(sig: DeviceSignals): Quality {
  // Mobile / touch devices: cap by core count.
  if (sig.coarsePointer) {
    if (sig.cores <= 4 || (sig.deviceMemory ?? 8) <= 3) return "low";
    return "med";
  }
  // Desktop with a pointer.
  if (sig.cores <= 4) return "med";
  return "high";
}

export function readDeviceSignals(): DeviceSignals {
  const nav = navigator as Navigator & { deviceMemory?: number };
  return {
    cores: nav.hardwareConcurrency || 4,
    deviceMemory: nav.deviceMemory,
    coarsePointer: window.matchMedia("(pointer: coarse)").matches,
    dpr: window.devicePixelRatio || 1,
  };
}

export function useQualityTier(): Quality {
  const [tier, setTier] = useState<Quality>("high");
  useEffect(() => {
    setTier(computeQualityTier(readDeviceSignals()));
  }, []);
  return tier;
}
```

```ts
import * as THREE from "three";
import type { Vec3 } from "@/components/room/content/objects";
import type { StationId } from "@/components/room/state/room-store";

export function buildCurve(points: Vec3[]): THREE.CatmullRomCurve3 {
  return new THREE.CatmullRomCurve3(
    points.map((p) => new THREE.Vector3(p.x, p.y, p.z)),
    false,
    "catmullrom",
    0.5,
  );
}

export function sampleAt(curve: THREE.CatmullRomCurve3, t: number): THREE.Vector3 {
  const clamped = Math.min(1, Math.max(0, t));
  return curve.getPointAt(clamped);
}

/**
 * Arc-length fraction (0..1) of each control point along the curve, found by
 * nearest-sample search over an arc-length-spaced tessellation.
 */
export function computeControlOffsets(
  curve: THREE.CatmullRomCurve3,
  points: Vec3[],
  samples = 512,
): number[] {
  const spaced = curve.getSpacedPoints(samples); // samples+1 arc-equidistant points
  return points.map((p) => {
    let bestIdx = 0;
    let bestDist = Infinity;
    for (let i = 0; i <= samples; i++) {
      const dx = spaced[i].x - p.x;
      const dy = spaced[i].y - p.y;
      const dz = spaced[i].z - p.z;
      const d = dx * dx + dy * dy + dz * dz;
      if (d < bestDist) {
        bestDist = d;
        bestIdx = i;
      }
    }
    return bestIdx / samples;
  });
}

export interface OffsetEntry {
  id: StationId;
  offset: number;
}

export function nearestStation(
  offset: number,
  entries: OffsetEntry[],
  threshold = 0.08,
): StationId | null {
  let best: OffsetEntry | null = null;
  let bestDist = Infinity;
  for (const e of entries) {
    const d = Math.abs(e.offset - offset);
    if (d < bestDist) {
      bestDist = d;
      best = e;
    }
  }
  return best && bestDist <= threshold ? best.id : null;
}
```

```ts
  curve: THREE.CatmullRomCurve3;
  introEnd: number; // offset where the overhead dive-in ends (last intro keyframe)
  stationEntries: StationEntry[];

let cached: Tour | null = null;

/** Builds the tour spline once and derives station offsets from arc length. */
export function getTour(): Tour {
  if (cached) return cached;
  const curve = buildCurve(tourPoints);
  const offsets = computeControlOffsets(curve, tourPoints);
  cached = {

    introEnd: offsets[introPath.length - 1],
    stationEntries: stations.map((s, i) => ({

      label: s.label,
      offset: offsets[introPath.length + i],
      cameraPos: s.cameraPos,
      lookAt: s.lookAt,
    })),
  };
  return cached;
```

```ts
import * as THREE from "three";

/** Recursively dispose geometries/materials/textures under an object. */
export function disposeObject(root: THREE.Object3D): void {
  root.traverse((obj) => {
    const mesh = obj as THREE.Mesh;
    if (mesh.geometry) mesh.geometry.dispose();
    const mat = mesh.material;
    if (Array.isArray(mat)) mat.forEach(disposeMaterial);
    else if (mat) disposeMaterial(mat);
  });
}

function disposeMaterial(mat: THREE.Material): void {
  for (const key of Object.keys(mat)) {
    const value = (mat as unknown as Record<string, unknown>)[key];
    if (value && value instanceof THREE.Texture) value.dispose();
  }
  mat.dispose();
}
```

- [ ] **Step 1: Implement the camera rig** — `components/room/scene/camera-rig.tsx`

```tsx
"use no memo";
"use client";
import { useEffect, useRef } from "react";
import { useFrame, useThree } from "@react-three/fiber";
import { useScroll } from "@react-three/drei";
import * as THREE from "three";
import { sampleAt, nearestStation } from "@/components/room/lib/curve";
import { getTour } from "@/components/room/lib/tour";
import { lockScroll } from "@/components/room/lib/scroll-lock";
import { useRoomStore } from "@/components/room/state/room-store";

const ROOM_CENTER = new THREE.Vector3(0, 0.6, 0);

export function CameraRig() {
  const camera = useThree((s) => s.camera);
  const scroll = useScroll();
  const { curve, introEnd, stationEntries } = getTour();
  const lookTarget = useRef(new THREE.Vector3(0, 0.6, 0));
  const setScrollOffset = useRoomStore((s) => s.setScrollOffset);
  const setActiveStation = useRoomStore((s) => s.setActiveStation);
  const lastStation = useRef<string | null>(null);

  useFrame(() => {
    const offset = scroll.offset;
    const reducedMotion = useRoomStore.getState().reducedMotion;

    // position: raw offset everywhere so derived station offsets stay aligned
    const pos = sampleAt(curve, offset);
    camera.position.lerp(pos, reducedMotion ? 1 : 0.12);

    // look target
    const target = new THREE.Vector3();
    let nearId: ReturnType<typeof nearestStation> = null;
    if (offset < introEnd) {
      // dive-in: gaze pinned to the room center while descending overhead,
      // easing toward the first station's subject as the camera levels out
      const p = offset / introEnd;
      const first = stationEntries[0].lookAt;
      target
        .copy(ROOM_CENTER)
        .lerp(
          new THREE.Vector3(first.x, first.y, first.z),
          THREE.MathUtils.smoothstep(p, 0.6, 1),
        );
    } else {
      const ahead = sampleAt(curve, Math.min(1, offset + 0.05));
      target.copy(ahead);
      nearId = nearestStation(offset, stationEntries, 0.1);
      if (nearId) {
        const st = stationEntries.find((s) => s.id === nearId)!;
        target.set(st.lookAt.x, st.lookAt.y, st.lookAt.z);
      }
    }
    lookTarget.current.lerp(target, reducedMotion ? 1 : 0.1);
    camera.lookAt(lookTarget.current);

    // publish state (throttled to station changes + offset)
    setScrollOffset(offset);
    if (nearId !== lastStation.current) {
      lastStation.current = nearId;
      setActiveStation(nearId);
    }
  });

  return null;
}

/**
 * Applies section-nav jump requests to the scroll element and freezes the
 * tour scroll while a modal is open (so it resumes exactly where it froze).
 */
export function ScrollBridge() {
  const scroll = useScroll();
  const requested = useRoomStore((s) => s.requestedOffset);
  const clear = useRoomStore((s) => s.clearRequestedOffset);
  const openObject = useRoomStore((s) => s.openObject);
  const prev = useRef<number | null>(null);

  useEffect(() => {
    if (!openObject) return;
    return lockScroll(scroll.el); // restore runs on close/unmount
  }, [openObject, scroll]);

  useFrame(() => {
    if (requested !== null && requested !== prev.current) {
      prev.current = requested;
      const el = scroll.el;
      const top = requested * (el.scrollHeight - el.clientHeight);
      el.scrollTo({ top, behavior: "smooth" });
      clear();
    }
  });
  return null;
}
```

```tsx
"use no memo";
"use client";
import { useState, useCallback } from "react";
import { useGLTF, Html, Outlines } from "@react-three/drei";
import type { ThreeEvent } from "@react-three/fiber";
import { ASSETS, type AssetId } from "@/components/room/content/assets";
import type { InteractiveObject as IObj } from "@/components/room/content/objects";
import { useRoomStore } from "@/components/room/state/room-store";

export function cursorFor(hovered: boolean): "pointer" | "auto" {
  return hovered ? "pointer" : "auto";
}

/** Loads and renders a single GLB by AssetId at the origin of its parent group. */
export function GltfModel({ id }: { id: AssetId }) {
  const { scene } = useGLTF(ASSETS[id].file);
  // clone so the same GLB can be reused if referenced twice
  return <primitive object={scene.clone()} />;
}

function toTransform(t: IObj["transform"]) {
  return {
    position: [t.position.x, t.position.y, t.position.z] as [number, number, number],
    rotation: [0, t.rotationY ?? 0, 0] as [number, number, number],
    scale: t.scale ?? 1,
  };
}

export function InteractiveObject({ obj }: { obj: IObj }) {
  const [hovered, setHovered] = useState(false);
  const openObjectPanel = useRoomStore((s) => s.openObjectPanel);
  const toggleLighting = useRoomStore((s) => s.toggleLighting);
  const markVisited = useRoomStore((s) => s.markVisited);
  const reducedMotion = useRoomStore((s) => s.reducedMotion);
  const t = toTransform(obj.transform);

  const activate = useCallback(
    (e?: ThreeEvent<MouseEvent>) => {
      e?.stopPropagation();
      if (obj.panel) {
        openObjectPanel(obj.id);
      } else if (obj.action === "toggleLighting") {
        toggleLighting();
        markVisited(obj.id);
      } else if (obj.action === "spin") {
        markVisited(obj.id);
        // fan spin handled by store visited + ceilingfan component in Task 13
      }
    },
    [obj, openObjectPanel, toggleLighting, markVisited],
  );

  return (
    <group position={t.position} rotation={t.rotation} scale={t.scale}>
      <group
        onPointerOver={(e) => {
          e.stopPropagation();
          setHovered(true);
          document.body.style.cursor = "pointer";
        }}
        onPointerOut={() => {
          setHovered(false);
          document.body.style.cursor = "auto";
        }}
        onClick={activate}
      >
        {obj.models.map((m) => (
          <GltfModel key={m} id={m} />
        ))}
        {hovered && <Outlines thickness={4} color="#111" />}
      </group>

      {obj.hotspot && (
        <Html
          position={[
            obj.hotspot.x - obj.transform.position.x,
            obj.hotspot.y - obj.transform.position.y,
            obj.hotspot.z - obj.transform.position.z,
          ]}
          center
          distanceFactor={8}
          occlude
        >
          <button
            onClick={() => activate()}
            className="pointer-events-auto whitespace-nowrap border-2 border-border bg-card px-2 py-1 font-mono text-[10px] text-foreground shadow-[3px_3px_0px_0px_var(--border)] transition-transform hover:-translate-y-0.5"
            style={{ opacity: hovered ? 1 : 0.85, transition: reducedMotion ? "none" : undefined }}
          >
            {obj.label}
          </button>
        </Html>
      )}
    </group>
  );
}
```

```tsx
"use client";
import { useEffect } from "react";
import { useRoomStore } from "@/components/room/state/room-store";
import { interactiveObjects, type PanelKey } from "@/components/room/content/objects";
import { AboutPanel } from "./panels/about";
import { ProjectsOsPanel } from "./panels/projects-os";
import { GalleryPanel } from "./panels/gallery";
import { HobbiesPanel } from "./panels/hobbies";
import { NowPanel } from "./panels/now";
import { ContactPanel } from "./panels/contact";
import { CreditsPanel } from "./panels/credits";

const PANELS: Record<PanelKey, () => React.ReactElement> = {
  about: AboutPanel,
  "projects-os": ProjectsOsPanel,
  gallery: GalleryPanel,
  hobbies: HobbiesPanel,
  now: NowPanel,
  contact: ContactPanel,
  credits: CreditsPanel,
};

const TITLES: Record<PanelKey, string> = {
  about: "About me",
  "projects-os": "Projects",
  gallery: "Photos",
  hobbies: "Outside of code",
  now: "Right now",
  contact: "Say hello",
  credits: "Credits",
};

export function ObjectOverlay() {
  const openObject = useRoomStore((s) => s.openObject);
  const close = useRoomStore((s) => s.closeObjectPanel);

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => e.key === "Escape" && close();
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [close]);

  if (!openObject) return null;

  // credits is opened via the special "credits" object id (see HUD); map others via registry
  const panelKey: PanelKey | undefined =
    openObject === "credits"
      ? "credits"
      : interactiveObjects.find((o) => o.id === openObject)?.panel;
  if (!panelKey) return null;

  const Panel = PANELS[panelKey];
  return (
    <div className="fixed inset-0 z-30 grid place-items-center bg-black/40 p-4" onClick={close}>
      <div
        role="dialog"
        aria-label={TITLES[panelKey]}
        onClick={(e) => e.stopPropagation()}
        className="max-h-[80vh] w-full max-w-md overflow-y-auto border-2 border-border bg-card p-6 text-card-foreground shadow-[6px_6px_0px_0px_var(--border)]"
      >
        <div className="mb-4 flex items-center justify-between">
          <h2 className="text-xl font-bold">{TITLES[panelKey]}</h2>
          <button
            onClick={close}
            aria-label="Close"
            className="border-2 border-border bg-background px-2 py-0.5 font-mono text-sm shadow-[2px_2px_0px_0px_var(--border)]"
          >
            ✕
          </button>
        </div>
        <Panel />
      </div>
    </div>
  );
}
```
